\subsection{Chapter 13: The Average Causal Effect on the Treated Units and Other Estimands}


\begin{definition}[CATE, ATT, and ATE]\label{13:def:CATE-ATT}
The conditional average treatment effect (CATE) is $\tau(X) = E\{ Y(1)-Y(0) \mid X \}$.
The average treatment effect on the treated (ATT) is $\tau_\mathrm{T} = E\{ Y(1)-Y(0) \mid Z=1 \}$.
The average treatment effect on the control (ATC) is $\tau_\mathrm{C} = E\{ Y(1)-Y(0) \mid Z=0 \}$.
The average treatment effect (ATE) is $\tau = E\{ Y(1)-Y(0) \}$.
\end{definition}
 \Paragraph{13.1~Comparing $\tau_\mathrm{T}, \tau_\mathrm{C}$, and $\tau$}\label{hw::difference-tau-T-tau}

Assume $Z\ind \{Y(1), Y(0)\} \mid X$. Recall $e(X) = \pr(Z=1\mid X)$ is the propensity score, $e = \pr(Z=1)$ is the marginal probability of the treatment, and $\tau(X) = E\{Y(1)-Y(0) \mid X\}$ is the CATE.
Show that
$$
\tau_\mathrm{T} - \tau = \frac{\cov\{e(X), \tau(X)\}}{e},\quad
\tau_\mathrm{C} - \tau = - \frac{\cov\{e(X), \tau(X)\}}{1-e}.
$$

Remark: The results above also imply that $\tau_\mathrm{T} - \tau_\mathrm{C} = \cov\{e(X), \tau(X)\} / \{e(1-e)\}$. So the differences among $\tau_\mathrm{T}, \tau_\mathrm{C}, \tau$ depend on the covariance between the propensity score and CATE. When $e(X)$ and $\tau(X)$ are uncorrelated, their differences are 0.

\Paragraph{Solution to 13.1}

Under the ignorability assumption $Z \ind \{Y(1), Y(0)\} \mid X$, the conditional average treatment effect satisfies:
\begin{equation*}
    \tau(X) = E\{Y(1) - Y(0) \mid X\} = E\{Y(1) - Y(0) \mid Z=1, X\}.
\end{equation*}

\textbf{1. Derivation for $\tau_\mathrm{T} - \tau$:}

By the Law of Iterated Expectations, the average treatment effect on the treated (ATT) is:
\begin{align*}
    \tau_\mathrm{T} &= E\{Y(1) - Y(0) \mid Z=1\} \\
    &= E[ E\{Y(1) - Y(0) \mid Z=1, X\} \mid Z=1 ] \\
    &= E\{ \tau(X) \mid Z=1 \}.
\end{align*}
Using the property of conditional expectation $E(A \mid B) = E(A \cdot \mathbbm{I}_B) / \pr(B)$:
\begin{equation}
    \tau_\mathrm{T} = \frac{E\{ \tau(X) \mathbbm{I}(Z=1) \}}{\pr(Z=1)} = \frac{E[ E\{ \tau(X) \mathbbm{I}(Z=1) \mid X \} ]}{e} = \frac{E\{ \tau(X) e(X) \}}{e}. \label{eq:tauT_step}
\end{equation}
Recall the definition of covariance: $\cov\{e(X), \tau(X)\} = E\{e(X) \tau(X)\} - E\{e(X)\}E\{\tau(X)\}$. Since $E\{e(X)\} = e$ and $E\{\tau(X)\} = \tau$, we have:
\begin{equation*}
    E\{e(X) \tau(X)\} = \cov\{e(X), \tau(X)\} + e\tau.
\end{equation*}
Substituting this into \eqref{eq:tauT_step}:
\begin{equation*}
    \tau_\mathrm{T} = \frac{\cov\{e(X), \tau(X)\} + e\tau}{e} = \frac{\cov\{e(X), \tau(X)\}}{e} + \tau.
\end{equation*}
Thus, $\tau_\mathrm{T} - \tau = \frac{\cov\{e(X), \tau(X)\}}{e}$.

\vspace{1em}
\textbf{2. Derivation for $\tau_\mathrm{C} - \tau$:}

Similarly, for the average treatment effect on the control (ATC):
\begin{align*}
    \tau_\mathrm{C} &= E\{ \tau(X) \mid Z=0 \} = \frac{E\{ \tau(X) \mathbbm{I}(Z=0) \}}{\pr(Z=0)} \\
    &= \frac{E\{ \tau(X) (1 - e(X)) \}}{1-e} \\
    &= \frac{E\{\tau(X)\} - E\{\tau(X)e(X)\}}{1-e}.
\end{align*}
Substituting $E\{\tau(X)\} = \tau$ and $E\{\tau(X)e(X)\} = \cov\{e(X), \tau(X)\} + e\tau$:
\begin{align*}
    \tau_\mathrm{C} &= \frac{\tau - [\cov\{e(X), \tau(X)\} + e\tau]}{1-e} \\
    &= \frac{\tau(1-e) - \cov\{e(X), \tau(X)\}}{1-e} \\
    &= \tau - \frac{\cov\{e(X), \tau(X)\}}{1-e}.
\end{align*}
Thus, $\tau_\mathrm{C} - \tau = - \frac{\cov\{e(X), \tau(X)\}}{1-e}$.

 \Paragraph{13.6~Estimating individual effect and conditional average causal effect}\label{hw::individual-effect-prediction}

Assume that $\{Z_i, X_i, Y_i(1), Y_i(0)\}_{i=1}^n \iidsim \{Z, X, Y(1), Y(0)\}$. The individual effect is $\tau_i = Y_i(1) - Y_i(0)$ and the conditional average causal effect is $\tau(X_i) = E\{Y_i(1) - Y_i(0) \mid X_i\}$.

\begin{enumerate}
\item
Under randomization with $Z_i\ind \{Y_i(1), Y_i(0)\}$ and $e = \pr(Z_i = 1)$, show that
$$
\delta_i = \frac{Z_i Y_i}{e} - \frac{(1-Z_i)Y_i}{1-e}
$$
is an unbiased predictor of the individual effect in the sense that
$$
E(\delta_i - \tau_i) = 0\quad (i=1,\ldots, n).
$$
Further show that $E(\delta_i) = \tau$ for all $i=1,\ldots, n$.

\item
Under ignorability with $Z_i\ind \{Y_i(1), Y_i(0)\}\mid X_i$ and $e(X_i) = \pr(Z_i=1\mid X_i)$, show that
$$
\delta_i = \frac{Z_i Y_i}{e(X_i)} - \frac{(1-Z_i)Y_i}{1-e(X_i)}
$$
is an unbiased predictor of the individual effect and the conditional average causal effect in the sense that
$$
E(\delta_i - \tau_i) = 0, \quad
E\{\delta_i - \tau(X_i)\} = 0, \quad (i=1,\ldots, n).
$$
Further show that $E(\delta_i) = \tau$ for all $i=1,\ldots, n$.
\end{enumerate}

\Paragraph{Solution to 13.6}

First, note that the observed outcome is $Y_i = Z_i Y_i(1) + (1-Z_i) Y_i(0)$. Because $Z_i$ is binary, $Z_i^2 = Z_i$ and $Z_i(1-Z_i) = 0$. Thus, $Z_i Y_i = Z_i Y_i(1)$ and $(1-Z_i) Y_i = (1-Z_i) Y_i(0)$.

\begin{enumerate}
\item \textbf{Under Randomization:}
The predictor is $\delta_i = \frac{Z_i Y_i(1)}{e} - \frac{(1-Z_i) Y_i(0)}{1-e}$. Conditioning on the potential outcomes:
\begin{align*}
    E(\delta_i \mid Y_i(1), Y_i(0)) &= \frac{Y_i(1)}{e} E(Z_i \mid Y_i(1), Y_i(0)) - \frac{Y_i(0)}{1-e} E(1-Z_i \mid Y_i(1), Y_i(0)) \\
    &= \frac{Y_i(1)}{e} \cdot e - \frac{Y_i(0)}{1-e} \cdot (1-e) \\
    &= Y_i(1) - Y_i(0) = \tau_i,
\end{align*}
where we used randomization $Z_i \ind \{Y_i(1), Y_i(0)\}$, implying $E(Z_i \mid Y_i(1), Y_i(0)) = E(Z_i) = e$.
By the Law of Iterated Expectations (LIE):
\begin{equation*}
    E(\delta_i - \tau_i) = E[ E(\delta_i - \tau_i \mid Y_i(1), Y_i(0)) ] = E[\tau_i - \tau_i] = 0.
\end{equation*}
Finally, $E(\delta_i) = E(\tau_i) = \tau$.

\item \textbf{Under Ignorability:}
The predictor is $\delta_i = \frac{Z_i Y_i(1)}{e(X_i)} - \frac{(1-Z_i) Y_i(0)}{1-e(X_i)}$. Conditioning on $(X_i, Y_i(1), Y_i(0))$:
\begin{align*}
    E(\delta_i \mid X_i, Y_i(1), Y_i(0)) &= \frac{Y_i(1)}{e(X_i)} E(Z_i \mid X_i, Y_i(1), Y_i(0)) - \frac{Y_i(0)}{1-e(X_i)} E(1-Z_i \mid X_i, Y_i(1), Y_i(0)) \\
    &= \frac{Y_i(1)}{e(X_i)} \cdot e(X_i) - \frac{Y_i(0)}{1-e(X_i)} \cdot (1-e(X_i)) \\
    &= \tau_i,
\end{align*}
where we used ignorability $Z_i \ind \{Y_i(1), Y_i(0)\} \mid X_i$. 
As before, $E(\delta_i - \tau_i) = E[ E(\delta_i - \tau_i \mid X_i, Y_i(1), Y_i(0)) ] = 0$.
For the CATE $\tau(X_i)$, by LIE:
\begin{equation*}
    E(\delta_i \mid X_i) = E[ E(\delta_i \mid X_i, Y_i(1), Y_i(0)) \mid X_i ] = E(\tau_i \mid X_i) = \tau(X_i).
\end{equation*}
Thus, $E(\delta_i - \tau(X_i)) = E[ E(\delta_i - \tau(X_i) \mid X_i) ] = 0$.
Finally, $E(\delta_i) = E[E(\delta_i \mid X_i)] = E[\tau(X_i)] = \tau$.
\end{enumerate}


\begin{definition}[General estimand $\tau^h$]\label{13:def:tau-h}
For a given function $h(X)$, the general estimand is
$$
\tau^h = \frac{E\{h(X)\tau(X)\}}{E\{h(X)\}} = \frac{E[h(X)\{\mu_1(X) - \mu_0(X)\}]}{E\{h(X)\}},
$$
where $\mu_1(X) = E\{Y(1)\mid X\}$ and $\mu_0(X) = E\{Y(0)\mid X\}$.
\end{definition}

\begin{definition}[Doubly robust estimator]\label{13:def:DR-estimator}
The doubly robust estimator uses outcome regression models $\mu_1(X,\beta_1)$ and $\mu_0(X,\beta_0)$, and propensity score model $e(X,\alpha)$:
\begin{align*}
\tilde{\mu}_1^{h,\mathrm{dr}} &= E\left[\frac{Zh(X)\{Y-\mu_1(X,\beta_1)\}}{e(X,\alpha)} + h(X)\mu_1(X,\beta_1)\right], \\
\tilde{\mu}_0^{h,\mathrm{dr}} &= E\left[\frac{(1-Z)h(X)\{Y-\mu_0(X,\beta_0)\}}{1-e(X,\alpha)} + h(X)\mu_0(X,\beta_0)\right].
\end{align*}
\end{definition}
 \Paragraph{13.10~Doubly robust estimation for general estimand}\label{hw::general-estimand-dr}

For a given $h(X)$, we have the following formulas for constructing the doubly robust estimator for $\tau^h$:
\begin{align*}
\tilde{\mu}_1^{h,\mathrm{dr}} &= E\left[\frac{Zh(X)\{ Y-\mu_1(X,\beta_1) \}}{e(X,\alpha)} + h(X)\mu_1(X,\beta_1)\right], \\
\tilde{\mu}_0^{h,\mathrm{dr}} &= E\left[\frac{(1-Z)h(X)\{ Y-\mu_0(X,\beta_0) \}}{1-e(X,\alpha)} + h(X)\mu_0(X,\beta_0)\right].
\end{align*}
Show that under ignorability and overlap:
\begin{enumerate}
\item
if either $e(X,\alpha) = e(X)$ or $\mu_1(X,\beta_1) = \mu_1(X)$, then $\tilde{\mu}_1^{h,\mathrm{dr}} = E\{h(X)Y(1)\}$;
\item
if either $e(X,\alpha) = e(X)$ or $\mu_0(X,\beta_0) = \mu_0(X)$, then $\tilde{\mu}_0^{h,\mathrm{dr}} = E\{h(X)Y(0)\}$;
\item
if either $e(X,\alpha) = e(X)$ or $\{\mu_1(X,\beta_1) = \mu_1(X), \mu_0(X,\beta_0) = \mu_0(X)\}$, then
$$
\frac{\tilde{\mu}_1^{h,\mathrm{dr}} - \tilde{\mu}_0^{h,\mathrm{dr}}}{E\{h(X)\}} = \tau^h.
$$
\end{enumerate}

\Paragraph{Solution to 13.10}

Let $e^*(X) = e(X, \alpha)$ and $\mu_1^*(X) = \mu_1(X, \beta_1)$. We analyze $\tilde{\mu}_1^{h,\mathrm{dr}}$ first.
\begin{align*}
    \tilde{\mu}_1^{h,\mathrm{dr}} &= E\left[\frac{Zh(X)\{Y - \mu_1^*(X)\}}{e^*(X)} + h(X)\mu_1^*(X)\right] \\
    &= E\left[ \frac{Zh(X)Y(1)}{e^*(X)} - \frac{Zh(X)\mu_1^*(X)}{e^*(X)} + h(X)\mu_1^*(X) \right] \quad (\text{since } ZY = ZY(1)).
\end{align*}
Applying the Law of Iterated Expectations by conditioning on $X$:
\begin{align*}
    E[\tilde{\mu}_1^{h,\mathrm{dr}}] &= E\left[ E\left\{ \frac{Zh(X)Y(1)}{e^*(X)} - \frac{Zh(X)\mu_1^*(X)}{e^*(X)} + h(X)\mu_1^*(X) \mid X \right\} \right] \\
    &= E\left[ \frac{h(X) \mu_1(X) e(X)}{e^*(X)} - \frac{h(X)\mu_1^*(X)e(X)}{e^*(X)} + h(X)\mu_1^*(X) \right] \quad (\text{by } Z \ind Y(1) \mid X) \\
    &= E\left[ \frac{h(X)e(X)}{e^*(X)} \{ \mu_1(X) - \mu_1^*(X) \} + h(X)\mu_1^*(X) \right].
\end{align*}

\begin{enumerate}
    \item \textbf{Part 1:} 
    \begin{itemize}
        \item If $e^*(X) = e(X)$, then $E[\tilde{\mu}_1^{h,\mathrm{dr}}] = E[ h(X)\mu_1(X) - h(X)\mu_1^*(X) + h(X)\mu_1^*(X) ] = E[h(X)\mu_1(X)]$.
        \item If $\mu_1^*(X) = \mu_1(X)$, then $E[\tilde{\mu}_1^{h,\mathrm{dr}}] = E[ 0 + h(X)\mu_1(X) ] = E[h(X)\mu_1(X)]$.
    \end{itemize}
    In both cases, $E[\tilde{\mu}_1^{h,\mathrm{dr}}] = E[h(X)E\{Y(1)\mid X\}] = E[h(X)Y(1)]$.

    \item \textbf{Part 2:}
    Symmetrically, for $\tilde{\mu}_0^{h,\mathrm{dr}} = E\left[ \frac{(1-Z)h(X)\{Y - \mu_0^*(X)\}}{1 - e^*(X)} + h(X)\mu_0^*(X) \right]$:
    \begin{equation*}
        E[\tilde{\mu}_0^{h,\mathrm{dr}}] = E\left[ \frac{h(X)(1-e(X))}{1-e^*(X)} \{ \mu_0(X) - \mu_0^*(X) \} + h(X)\mu_0^*(X) \right].
    \end{equation*}
    If either $e^*(X) = e(X)$ or $\mu_0^*(X) = \mu_0(X)$, we have $E[\tilde{\mu}_0^{h,\mathrm{dr}}] = E[h(X)\mu_0(X)] = E[h(X)Y(0)]$.

    \item \textbf{Part 3:}
    If the conditions in Part 1 and Part 2 are met (which is true if either $e(X, \alpha) = e(X)$ or both outcome models are correct), then:
    \begin{equation*}
        \frac{E[\tilde{\mu}_1^{h,\mathrm{dr}}] - E[\tilde{\mu}_0^{h,\mathrm{dr}}]}{E[h(X)]} = \frac{E[h(X)Y(1)] - E[h(X)Y(0)]}{E[h(X)]} = \frac{E[h(X)\{Y(1)-Y(0)\}]}{E[h(X)]} = \tau^h.
    \end{equation*}
\end{enumerate}

\subsection{Chapter 14: Using the Propensity Score in Regressions for Causal Effects}

 \Paragraph{14.3~Weighted logistic regression with a binary outcome}\label{hw::logistic-regression}

With a binary outcome, we can replace linear outcome models with logistic outcome models. Show that with weights in the logistic regressions, the doubly robust estimators equal the outcome regression estimator. The result holds for both $\tau$ and $\tau_\mathrm{T}$.

\Paragraph{Solution to 14.3}

We show the result for the treated group component of $\tau$; the control group and the case for $\tau_\mathrm{T}$ follow the same logic. 

The outcome regression estimator for $E[Y(1)]$ is $\hat{\mu}_1^{\mathrm{or}} = n^{-1} \sum_{i=1}^n \mu_1(X_i, \hat{\beta}_1)$. 
The doubly robust estimator is:
\begin{equation*}
    \hat{\mu}_1^{\mathrm{dr}} = n^{-1} \sum_{i=1}^n \left[ \frac{Z_i Y_i}{e(X_i)} - \frac{Z_i - e(X_i)}{e(X_i)} \mu_1(X_i, \hat{\beta}_1) \right].
\end{equation*}
Rearranging the DR estimator, we have:
\begin{equation*}
    \hat{\mu}_1^{\mathrm{dr}} = n^{-1} \sum_{i=1}^n \mu_1(X_i, \hat{\beta}_1) + n^{-1} \sum_{i=1}^n \frac{Z_i}{e(X_i)} \{ Y_i - \mu_1(X_i, \hat{\beta}_1) \}.
\end{equation*}
Thus, $\hat{\mu}_1^{\mathrm{dr}} = \hat{\mu}_1^{\mathrm{or}}$ if and only if $\sum_{i=1}^n \frac{Z_i}{e(X_i)} \{ Y_i - \mu_1(X_i, \hat{\beta}_1) \} = 0$.

In a weighted logistic regression with weights $w_i = 1/e(X_i)$ for the treated group, the score equation for the intercept is:
\begin{equation*}
    \sum_{i=1}^n w_i Z_i \{ Y_i - \mu_1(X_i, \beta_1) \} = 0 \implies \sum_{i=1}^n \frac{Z_i}{e(X_i)} \{ Y_i - \mu_1(X_i, \beta_1) \} = 0.
\end{equation*}
Since $\hat{\beta}_1$ is the solution to this score equation, the condition is satisfied, and $\hat{\mu}_1^{\mathrm{dr}} = \hat{\mu}_1^{\mathrm{or}}$.

For $\tau_\mathrm{T}$, the weights in the logistic regression are $w_i = 1$ for $Z_i=1$, and the DR estimator uses different weights. The ATT outcome regression estimator is $n_1^{-1} \sum_{i: Z_i=1} Y_i$ (if we just use the treated units) or $n_1^{-1} \sum_{i: Z_i=1} \mu_1(X_i)$ if we use the whole sample to fit. Actually, for $\tau_\mathrm{T}$, the DR estimator is:
\begin{equation*}
    \hat{\tau}_{\mathrm{T, dr}} = \frac{1}{n_1} \sum_{i=1}^n Z_i Y_i - \frac{1}{n_1} \sum_{i=1}^n \frac{e(X_i)(1-Z_i)}{1-e(X_i)} \{ Y_i - \mu_0(X_i, \hat{\beta}_0) \} - \frac{1}{n_1} \sum_{i=1}^n Z_i \mu_0(X_i, \hat{\beta}_0).
\end{equation*}
Weighted logistic regression for the control group with weights $w_i = e(X_i)/(1-e(X_i))$ ensures $\sum \frac{e(X_i)(1-Z_i)}{1-e(X_i)} (Y_i - \mu_0(X_i)) = 0$, leading to $\hat{\tau}_{\mathrm{T, dr}} = \hat{\tau}_{\mathrm{T, or}}$.

\subsection{Chapter 17: E-Value: Evidence for Causation in Observational Studies with Unmeasured Confounding}

\begin{definition}[Risk ratios and risk difference]\label{17:def:RR-RD}
For binary $Y$, the true conditional causal risk ratio is
$$\RR_{ZY\mid x}^\true = \frac{\pr\{Y(1)=1\mid X=x\}}{\pr\{Y(0)=1\mid X=x\}}.$$
The observed conditional risk ratio is
$$\RR_{ZY\mid x}^\obs = \frac{\pr(Y=1\mid Z=1,X=x)}{\pr(Y=1\mid Z=0,X=x)}.$$
The treatment-confounder risk ratio is
$$\RR_{ZU\mid x} = \frac{\pr(U=1\mid Z=1,X=x)}{\pr(U=1\mid Z=0,X=x)}.$$
The confounder-outcome risk ratio is
$$\RR_{UY\mid x} = \frac{\pr(Y=1\mid U=1,X=x)}{\pr(Y=1\mid U=0,X=x)}.$$
The risk difference is $\RD_{ZY} = \pr(Y=1\mid Z=1) - \pr(Y=1\mid Z=0)$.
\end{definition}

\begin{thesis}\label{17.1}\emph{(Theorem 17.1, Ding, 2025)}
Under $Z\ind Y\mid (X, U)$, assume
\begin{equation}\label{eq::evalue-17.1}
\RR_{ZY\mid x}^\obs > 1, \quad \RR_{ZU\mid x} > 1,\quad \RR_{UY\mid x} > 1.
\end{equation}
We have
$$
\RR_{ZY\mid x}^\obs \leq \frac{ \RR_{ZU\mid x}  \RR_{UY\mid x} }{  \RR_{ZU\mid x} + \RR_{UY\mid x} - 1} .
$$
\end{thesis}

\begin{myproof}{Theorem}{\ref{17.1}}
Define 
$$ 
f_{1,x} = \pr(U=1\mid Z=1,X=x),\quad
f_{0,x}  = \pr(U=1\mid Z=0,X=x) .
$$ 
We can decompose $\RR_{ZY\mid x}^\obs$ as
\begin{align*}
\RR_{ZY\mid x}^\obs  
&= \frac{  \pr( Y  = 1\mid Z=1, X=x  ) }{ \pr( Y  = 1\mid  Z=0,  X=x  ) } \\
&= \frac{ \pr(U=1\mid Z=1,  X=x) \pr( Y  = 1\mid Z=1, U=1,  X=x  ) + \pr(U=0\mid Z=1,  X=x) \pr( Y  = 1\mid Z=1, U=0,  X=x  ) }
{  \pr(U=1\mid Z=0,  X=x) \pr( Y  = 1\mid  Z=0, U=1,  X=x  )  +  \pr(U=0\mid Z=0,  X=x) \pr( Y  = 1\mid  Z=0, U=0,  X=x  )   } \\
&= \frac{ \pr(U=1\mid Z=1,  X=x) \pr( Y  = 1\mid U=1,  X=x  ) + \pr(U=0\mid Z=1,  X=x) \pr( Y  = 1\mid U=0,  X=x  )  }
{   \pr(U=1\mid Z=0,  X=x) \pr( Y  = 1\mid   U=1,  X=x  )  +  \pr(U=0\mid Z=0,  X=x) \pr( Y  = 1\mid  U=0,  X=x  )   } \\
&= \frac{  f_{1,x} \RR_{UY\mid x} + 1- f_{1,x}  }{ f_{0,x} \RR_{UY\mid x} + 1- f_{0,x}  } \\
&= \frac{  (\RR_{UY\mid x} - 1) f_{1,x} + 1  }{      \frac{  \RR_{UY\mid x} - 1}{  \RR_{ZU\mid x} }    f_{1,x} + 1  } .
\end{align*}
By the technical lemma (Exercise 17.1), $f(x) = \frac{ax+1}{bx+1}$ is increasing in $x$ if $a > b$. Here $a = \RR_{UY\mid x} - 1$ and $b = \frac{\RR_{UY\mid x} - 1}{\RR_{ZU\mid x}}$. Since $\RR_{ZU\mid x} > 1$, we have $a > b$. Thus, $\RR_{ZY\mid x}^\obs$ is increasing in $f_{1,x}$. Letting $f_{1,x} = 1$, we obtain the upper bound:
$$
\RR_{ZY\mid x}^\obs \leq \frac{  (\RR_{UY\mid x} - 1) + 1  }{      \frac{  \RR_{UY\mid x} - 1}{  \RR_{ZU\mid x} }    + 1  }
= \frac{ \RR_{ZU\mid x}  \RR_{UY\mid x} }{  \RR_{ZU\mid x} + \RR_{UY\mid x} - 1} .
$$
\end{myproof}

 \Paragraph{17.1~A technical lemma for the proof of \cref{17.1}}\label{hw::technical-lemma}

Show that $f(x) = \frac{ax+1}{bx+1}$ is increasing in $x$ if $a > b$ and decreasing in $x$ if $a < b$.

\Paragraph{Solution to 17.1}

Consider the derivative of $f(x)$ with respect to $x$:
\begin{equation*}
    f'(x) = \frac{a(bx+1) - b(ax+1)}{(bx+1)^2} = \frac{abx + a - abx - b}{(bx+1)^2} = \frac{a - b}{(bx+1)^2}.
\end{equation*}
Since the denominator $(bx+1)^2$ is always positive:
\begin{itemize}
    \item If $a > b$, then $f'(x) > 0$, so $f(x)$ is increasing in $x$.
    \item If $a < b$, then $f'(x) < 0$, so $f(x)$ is increasing in $x$.
\end{itemize}

 \Paragraph{17.2~Technical assumption for Theorem 17.1}\label{hw::technical-assumption}

Revisit the proof of \cref{17.1}. Assume $Z\ind Y \mid (X, U)$. Show that if $\RR_{ZU|x} > 1$ but $\RR_{UY|x} < 1$, then $\RR_{ZY|x}^{\obs} < 1$.

\Paragraph{Solution to 17.2}

Condition on $X=x$ implicitly. Let $q_z = \pr(U=1 \mid Z=z)$ and $p_u = \pr(Y=1 \mid U=u)$. By $Z \ind Y \mid U$, we have $\pr(Y=1 \mid Z=z, U=u) = \pr(Y=1 \mid U=u) = p_u$.
Using the Law of Total Probability:
\begin{align*}
    \pr(Y=1 \mid Z=z) &= \pr(Y=1 \mid Z=z, U=1)\pr(U=1 \mid Z=z) + \pr(Y=1 \mid Z=z, U=0)\pr(U=0 \mid Z=z) \\
    &= p_1 q_z + p_0 (1 - q_z) = p_0 + (p_1 - p_0)q_z.
\end{align*}
The observed risk ratio is $\RR_{ZY}^{\obs} = \frac{p_0 + (p_1 - p_0)q_1}{p_0 + (p_1 - p_0)q_0}$. 
Given:
\begin{itemize}
    \item $\RR_{ZU} > 1 \implies \frac{q_1}{q_0} > 1 \implies q_1 > q_0$ (assuming $q_0 > 0$).
    \item $\RR_{UY} < 1 \implies \frac{p_1}{p_0} < 1 \implies p_1 < p_0$.
\end{itemize}
Then $\pr(Y=1 \mid Z=1) - \pr(Y=1 \mid Z=0) = (p_1 - p_0)(q_1 - q_0)$.
Since $p_1 - p_0 < 0$ and $q_1 - q_0 > 0$, the product is negative. 
Thus $\pr(Y=1 \mid Z=1) < \pr(Y=1 \mid Z=0)$, which implies $\RR_{ZY}^{\obs} < 1$.

 \Paragraph{17.6~Cornfield-type inequalities for the risk difference}\label{hw::cornfield-rd}

Consider binary $Z, Y, U$, and condition on $X$ implicitly. Assume latent ignorability given $U$. Show that under $Z\ind Y \mid U$, we have
$$
\RD_{ZY}^{\obs} = \RD_{ZU} \times \RD_{UY}
$$
where $\RD_{ZY}^{\obs}$ is the observed risk difference of $Z$ on $Y$, and $\RD_{ZU}$ and $\RD_{UY}$ are the treatment-confounder and confounder-outcome risk differences, respectively.

\Paragraph{Solution to 17.6}

From the decomposition in the solution to 17.2:
\begin{align*}
    \pr(Y=1 \mid Z=1) &= p_0 + (p_1 - p_0)q_1, \\
    \pr(Y=1 \mid Z=0) &= p_0 + (p_1 - p_0)q_0.
\end{align*}
Subtracting the two equations:
\begin{equation*}
    \RD_{ZY}^{\obs} = \pr(Y=1 \mid Z=1) - \pr(Y=1 \mid Z=0) = (p_1 - p_0)(q_1 - q_0).
\end{equation*}
By definition:
\begin{itemize}
    \item $\RD_{UY} = \pr(Y=1 \mid U=1) - \pr(Y=1 \mid U=0) = p_1 - p_0$.
    \item $\RD_{ZU} = \pr(U=1 \mid Z=1) - \pr(U=1 \mid Z=0) = q_1 - q_0$.
\end{itemize}
Therefore, $\RD_{ZY}^{\obs} = \RD_{ZU} \times \RD_{UY}$.

Remark: Without loss of generality, assume that $\RD_{ZY}^{\obs}, \RD_{ZU}, \RD_{UY}$ are all positive. Then the above implies:
$$
\min(\RD_{ZU}, \RD_{UY}) \geq \RD_{ZY}^{\obs}
$$
and
$$
\max(\RD_{ZU}, \RD_{UY}) \geq \sqrt{\RD_{ZY}^{\obs}}.
$$

These are the Cornfield inequalities for the risk difference with a binary confounder.

\subsection{Chapter 20: Overlap in Observational Studies: Difficulties and Opportunities}

\begin{definition}[Overlap and propensity score]\label{20:def:overlap}
The propensity score is $e(X) = \pr(Z=1\mid X)$.
The overlap condition requires $0 < e(X) < 1$.
The marginal probability of treatment is $e = \pr(Z=1)$.
Let $\lambda_1$ and $\lambda_0$ be the maximum eigenvalues of $\cov(X\mid Z=1)$ and $\cov(X\mid Z=0)$, respectively.
The strict overlap condition requires $\eta \leq e(X) \leq 1-\eta$ for some $\eta \in (0,1/2)$.
\end{definition}


\begin{assumption}[strict overlap]\label{20.1}
$\eta \leq e(X) \leq 1-\eta$ for some $\eta \in (0,1/2).$
\end{assumption}

\begin{thesis}\label{20.2}
Assumption \ref{20.1} implies that $\eta \leq e \leq 1-\eta$ and
\begin{align}
& p^{-1} \sum_{k=1}^p \big | E(X_k\mid Z=1) - E(X_k\mid Z=0) \big | \leq p^{-1/2} C^{1/2} \{ e\lambda_1^{1/2} + (1-e) \lambda_0^{1/2} \},
\end{align}
where
$$
C = \frac{ (e-\eta)(1-e-\eta) }{ e^2(1-e)^2 \eta (1-\eta) }
$$
is a positive constant depending only on $(e, \eta)$, and $\lambda_1$ and $\lambda_0$ are the maximum eigenvalues of the covariance matrices $\cov(X\mid Z=1)$ and $\cov(X\mid Z=0)$, respectively.
\end{thesis}


\beginemark{Relevant computation formulas:}
\begin{enumerate}
\item \textbf{E-value bounding formula:} The upper bound for the observed risk ratio is
$$\RR_{ZY\mid x}^\obs \leq \frac{\RR_{ZU\mid x} \cdot \RR_{UY\mid x}}{\RR_{ZU\mid x} + \RR_{UY\mid x} - 1}.$$
When $\RR_{ZU\mid x} = \RR_{UY\mid x} = r$, the bound simplifies to $r$.
\item \textbf{Constant $C$ in overlap inequality:}
$$C = \frac{(e-\eta)(1-e-\eta)}{e^2(1-e)^2\eta(1-\eta)}.$$
For practical computation, note that $C \approx 1/(e(1-e))$ when $\eta$ is small.
\item \textbf{IPW estimator for general estimand:} $\tau^h = E\{w(X) \cdot ZY / e(X)\} / E\{h(X)\}$ where $w(X) = h(X) + \{1-e(X)\}/e(X) \cdot something$.
\item \textbf{Risk difference product formula:} $\RD_{ZY}^\obs = \RD_{ZU} \times \RD_{UY}$ for binary confounders.
\end{enumerate}
\endremark}

 \Paragraph{20.1~Implications of overlap}\label{hw::overlap-implications}

In Part III of this book, causal inference with observational studies relies on two critical assumptions: ignorability $Z\ind \{Y(1), Y(0)\} \mid X$ and overlap $0 < e(X) < 1$.

D'Amour et al. (2021) pointed out the tension between these two assumptions: typically, more covariates make the ignorability assumption more plausible, but more covariates make the overlap assumption less plausible because the treatment becomes more predictable given more covariates.

Many statistical analyses require a strict version of overlap (Assumption \ref{20.1}).

Show that Assumption \ref{20.1} implies $\eta \leq e \leq 1 - \eta$ and the inequality above.

\Paragraph{Solution to 20.1}

Assumption \ref{20.1} states that $\eta \leq e(X) \leq 1 - \eta$.
By the Law of Iterated Expectations, the marginal probability of treatment is:
\begin{equation*}
    e = \pr(Z=1) = E[ \pr(Z=1 \mid X) ] = E[ e(X) ].
\end{equation*}
Since $\eta \leq e(X) \leq 1 - \eta$ for all $X$, it follows that:
\begin{equation*}
    E[\eta] \leq E[e(X)] \leq E[1 - \eta] \implies \eta \leq e \leq 1 - \eta.
\end{equation*}
The second part regarding covariate balance follows from Theorem 20.1 in the textbook. The constant $C$ is derived by bounding the variance of the weights $Z/e(X)$ and $(1-Z)/(1-e(X))$ under strict overlap. Specifically, $\var(Z/e(X) \mid X) = e(X)(1-e(X))/e(X)^2 = (1-e(X))/e(X)$. Under strict overlap, this is bounded by $(1-\eta)/\eta$.

\section{Theoretical Exercises: Identification Formulas for Sensitivity Analysis}

Denote $\mu_z(X) = E(Y \mid Z=z, X)$ and $e(X) = \pr(Z=1 \mid X)$.
Suppose that
$$
\frac{E\{Y(1) \mid Z=1, X\}}{E\{Y(1) \mid Z=0, X\}} = \epsilon_1(X), \quad
\frac{E\{Y(0) \mid Z=1, X\}}{E\{Y(0) \mid Z=0, X\}} = \epsilon_0(X)
$$
with known $\epsilon_1(X)$ and $\epsilon_0(X)$.

 \Paragraph{(a)~Identification formulas}\label{hw::sensitivity-id}

Show that
$$
E\{Y(1) \mid Z=0\} = E\{\mu_1(X)/\epsilon_1(X) \mid Z=0\}, \quad
E\{Y(0) \mid Z=1\} = E\{\mu_0(X)\epsilon_0(X) \mid Z=1\}
$$
and hence
$$
\mathrm{ACE} = E\{ZY + (1-Z)\mu_1(X)/\epsilon_1(X)\} - E\{Z\mu_0(X)\epsilon_0(X) + (1-Z)Y\}.
$$

\Paragraph{Solution to (a)}

From the sensitivity parameter definition for $Y(1)$:
\begin{equation*}
    E\{Y(1) \mid Z=0, X\} = \frac{E\{Y(1) \mid Z=1, X\}}{\epsilon_1(X)} = \frac{E\{Y \mid Z=1, X\}}{\epsilon_1(X)} = \frac{\mu_1(X)}{\epsilon_1(X)}.
\end{equation*}
By the Law of Iterated Expectations:
\begin{equation*}
    E\{Y(1) \mid Z=0\} = E[ E\{Y(1) \mid Z=0, X\} \mid Z=0 ] = E\{ \mu_1(X)/\epsilon_1(X) \mid Z=0 \}.
\end{equation*}
Similarly, for $Y(0)$:
\begin{equation*}
    E\{Y(0) \mid Z=1, X\} = E\{Y(0) \mid Z=0, X\} \epsilon_0(X) = E\{Y \mid Z=0, X\} \epsilon_0(X) = \mu_0(X)\epsilon_0(X).
\end{equation*}
Thus, $E\{Y(0) \mid Z=1\} = E\{ \mu_0(X)\epsilon_0(X) \mid Z=1 \}$.
The average causal effect (ACE) is $E[Y(1)] - E[Y(0)]$:
\begin{align*}
    E[Y(1)] &= E[Y(1) \mid Z=1] \pr(Z=1) + E[Y(1) \mid Z=0] \pr(Z=0) \\
    &= E[ ZY + (1-Z) E\{Y(1) \mid Z=0, X\} ] = E[ ZY + (1-Z) \mu_1(X)/\epsilon_1(X) ].
\end{align*}
By symmetric logic for $E[Y(0)]$:
\begin{equation*}
    E[Y(0)] = E[ Z E\{Y(0) \mid Z=1, X\} + (1-Z)Y ] = E[ Z \mu_0(X)\epsilon_0(X) + (1-Z)Y ].
\end{equation*}
Subtracting gives the required ACE formula.

 \Paragraph{(b)~IPW representation}\label{hw::sensitivity-ipw}

Show that
$$
E\{Y(1)\} = E\left\{w_1(X)\frac{ZY}{e(X)}\right\}, \quad
E\{Y(0)\} = E\left\{w_0(X)\frac{(1-Z)Y}{1-e(X)}\right\},
$$
where $w_1(X) = e(X) + \{1-e(X)\}/\epsilon_1(X)$ and $w_0(X) = e(X)\epsilon_0(X) + \{1-e(X)\}$.

\Paragraph{Solution to (b)}

From Part (a), we have $E[Y(1) \mid X] = \pr(Z=1 \mid X) E[Y(1) \mid Z=1, X] + \pr(Z=0 \mid X) E[Y(1) \mid Z=0, X]$.
Substituting the sensitivity relationship:
\begin{align*}
    E[Y(1) \mid X] &= e(X) \mu_1(X) + \{1-e(X)\} \frac{\mu_1(X)}{\epsilon_1(X)} \\
    &= \mu_1(X) \left[ e(X) + \frac{1-e(X)}{\epsilon_1(X)} \right] = \mu_1(X) w_1(X).
\end{align*}
Since $\mu_1(X) = E[ZY/e(X) \mid X]$, we have:
\begin{equation*}
    E[Y(1)] = E[ \mu_1(X) w_1(X) ] = E[ E\{ZY/e(X) \mid X\} w_1(X) ] = E\left\{ w_1(X) \frac{ZY}{e(X)} \right\}.
\end{equation*}
By symmetry, $E[Y(0) \mid X] = e(X) \mu_0(X)\epsilon_0(X) + \{1-e(X)\} \mu_0(X) = \mu_0(X) w_0(X)$, which yields the IPW formula for $Y(0)$.

 \Paragraph{(c)~Doubly robust estimation (Optional)}\label{hw::sensitivity-dr}

Derive the doubly robust estimation formula.

\Paragraph{Solution to (c)}

The doubly robust estimator for $E[Y(1)]$ combines the OR and IPW components:
\begin{equation*}
    \tilde{\mu}_1^{\mathrm{dr}} = E\left[ \frac{ZY w_1(X)}{e(X)} - \frac{Z - e(X)}{e(X)} \mu_1(X) w_1(X) \right].
\end{equation*}
Expanding this:
\begin{equation*}
    \tilde{\mu}_1^{\mathrm{dr}} = E\left[ \frac{Z w_1(X)}{e(X)} \{ Y - \mu_1(X) \} + \mu_1(X) w_1(X) \right].
\end{equation*}
Similarly, for $E[Y(0)]$:
\begin{equation*}
    \tilde{\mu}_0^{\mathrm{dr}} = E\left[ \frac{(1-Z) w_0(X)}{1 - e(X)} \{ Y - \mu_0(X) \} + \mu_0(X) w_0(X) \right].
\end{equation*}
The DR estimator for ACE is then $\tilde{\mu}_1^{\mathrm{dr}} - \tilde{\mu}_0^{\mathrm{dr}}$.
