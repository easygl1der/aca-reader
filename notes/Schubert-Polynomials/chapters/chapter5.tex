\chapter{Quantum Double Schubert Polynomials Positivity Conjecture}
\label{chap:QuantumDoublePositivityConjecture}

\section{From Classical to Quantum: Why Extend?}

In \cref{chap:EquivariantQuantumPositivity}（见 \cite{Mihalcea}），Mihalcea proved the positivity of structure constants for two Schubert classes in equivariant quantum cohomology:

\[
\sigma(u) \circ \sigma(v) = \sum_{d} \sum_{w} q^d c_{u,v}^{w,d} \sigma(w)
\]

where the coefficients $c_{u,v}^{w,d} \in \Lambda = \mathbb{Z}[x_1, \ldots, x_r]$ are polynomials with nonnegative coefficients.

The natural next question: \textbf{Can we establish a similar positivity result for double Schubert polynomials in the quantum setting?}

\section{The Quantum Double Schubert Positivity Conjecture}

\begin{Conjecture}[Quantum Double Schubert Positivity]
\label{conj:QuantumDoublePositivity}
For $u, v, w \in S_{\infty}$, the product of quantum double Schubert polynomials expands as:

\[
\mathfrak{S}_u^{(q)}(\mathbf{x}; \mathbf{y}) \cdot \mathfrak{S}_v^{(q)}(\mathbf{x}; \mathbf{z}) = \sum_{w \in S_n} c_{u,v}^{w,(q)}(\mathbf{y}; \mathbf{z}; q) \cdot \mathfrak{S}_w^{(q)}(\mathbf{x}; \mathbf{y})
\]

The coefficients satisfy:
\[
c_{\mathbf{u},\mathbf{v}}^{\mathbf{w},(q)}(y; z; q) \in \mathbb{N}[y_i - z_j, q_k]_{i,j \geq 1, k \geq 1}
\]
namely, the coefficients are polynomials in $y_i - z_j$ and $q_k$ with nonnegative integer coefficients.
\end{Conjecture}

\textbf{Relationship with proved cases}:

\begin{enumerate}
\item \textbf{Double classical case} (Graham, 2001): For $q = 0$, the coefficients $c_{\mathbf{u},\mathbf{v}}^{\mathbf{w}}(y;z)$ are nonnegative polynomials in $t_i - y_j$.
\item \textbf{Equivariant quantum case} (\cite{Mihalcea}): For two Schubert classes, $c_{\mathbf{u},\mathbf{v}}^{\mathbf{w},d}$ are nonnegative polynomials in $x_i$.
\item \textbf{Quantum double case} (this conjecture): The unified setting with both $y_i - z_j$ parameters and quantum corrections $q_k$ remains open.
\end{enumerate}

The two key conjectures in this framework are:

\begin{Theorem}[Classical Double Schubert Positivity (Gao-Xiong, 2025) {\cite[Theorem 1.1]{GX2025}}]
\label{thm:ClassicalDoublePositivity}
For $u, v, w \in S_{\infty}$, the product of double Schubert polynomials satisfies:
\[
\mathfrak{S}_{\mathbf{u}}(\mathbf{x}; \mathbf{y}) \cdot \mathfrak{S}_{\mathbf{v}}(\mathbf{x}; \mathbf{z}) = \sum_{\mathbf{w} \in S_n} c_{\mathbf{u},\mathbf{v}}^{\mathbf{w}}(y; z) \cdot \mathfrak{S}_{\mathbf{w}}(\mathbf{x}; \mathbf{z})
\]
with coefficients $c_{\mathbf{u},\mathbf{v}}^{\mathbf{w}}(y; z) \in \mathbb{N}[y_i - z_j]_{i,j \geq 1}$.
\end{Theorem}

\begin{Conjecture}[Quantum Double Schubert Positivity]
\label{conj:QuantumDoubleSchubertPositivity}
For $u, v, w \in S_{\infty}$, the product of quantum double Schubert polynomials expands as:
\[
\mathfrak{S}_u^{(q)}(\mathbf{x}; \mathbf{y}) \cdot \mathfrak{S}_v^{(q)}(\mathbf{x}; \mathbf{z}) = \sum_{w \in S_n} c_{u,v}^{w,(q)}(\mathbf{y}; \mathbf{z}; q) \cdot \mathfrak{S}_w^{(q)}(\mathbf{x}; \mathbf{y})
\]
with coefficients satisfying:
\[
c_{\mathbf{u},\mathbf{v}}^{\mathbf{w},(q)}(y; z; q) \in \mathbb{N}[y_i - z_j, q_k]_{i,j \geq 1, k \geq 1}
\]
namely, polynomials in $y_i - z_j$ and quantum parameters $q_k$ with nonnegative integer coefficients.
\end{Conjecture}

\section{Geometric Interpretation}

The triple variables $(y_i - z_j, q_k)$ arise from three sources:

\begin{itemize}
\item $y_i$: records displacement from the first Schubert variety
\item $z_j$: records displacement from the second Schubert variety
\item $q_k$: records quantum corrections from curve counting (Gromov-Witten invariants)
\end{itemize}

The coefficient $c_{\mathbf{u},\mathbf{v}}^{\mathbf{w},(q)}(y;z;q)$ can be interpreted as counting curves meeting three Schubert varieties in the quantum setting.

\section{Current State}

This conjecture remains open. Progress includes:

\begin{itemize}
\item Mihalcea's proof of positivity for two Schubert classes in equivariant quantum cohomology \cite{Mihalcea}
\item Gao-Xiong's work establishing the framework for triple Schubert calculus \cite{GX2025}
\item The full quantum double Schubert positivity conjecture is a frontier topic in algebraic geometry and representation theory
\end{itemize}

\section{Remarks}

The purpose of this chapter is to record this conjecture for future reference. Proving this conjecture would unify the positivity phenomena in both classical and quantum Schubert calculus.
