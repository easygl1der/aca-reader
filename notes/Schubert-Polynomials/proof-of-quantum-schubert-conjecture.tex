% ABSTRACT
% ============================================================
\begin{abstract}
We prove the positivity conjecture for the product of quantum double Schubert polynomials with three sets of variables. Specifically, for any $u, v\in S_n$, the expansion coefficients
\[
c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})
\]
in
\[
\widetilde{\mathfrak{S}}_u(\mathbf{x};\mathbf{y})\cdot\widetilde{\mathfrak{S}}_v(\mathbf{x};\mathbf{z})
= \sum_d \sum_w q^d\, c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})\,
\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{z}).
\]
are polynomials in $\mathbf{y}$ and $\mathbf{z}$ with nonnegative integer coefficients in the negative roots $y_i - z_j$.
Our approach combines the equivariant quantum cohomology framework of Mihalcea \cite{Mihalcea} with the triple Schubert calculus of Gao--Xiong \cite{GX2025}, extending the Graham positivity theorem \cite{Gr} to the quantum setting.
\end{abstract}

% INTRODUCTION
% ============================================================
\section{Introduction}
\label{sec:introduction}

The theory of Schubert polynomials, initiated by Lascoux and Sch\"utzenberger, provides polynomial representatives for Schubert classes in the cohomology of flag varieties. In the classical setting, the Littlewood--Richardson coefficients $c_{u,v}^w$ appearing in
\[
\mathfrak{S}_u(\mathbf{x})\cdot\mathfrak{S}_v(\mathbf{x})
= \sum_w c_{u,v}^w\,\mathfrak{S}_w(\mathbf{x})
\label{eq:ClassicalProduct}
\]
are nonnegative integers, reflecting the geometric fact that they count intersection points of Schubert varieties. Graham's seminal theorem \cite{Gr} established that in the equivariant setting these coefficients become polynomials in the negative roots $x_i$ with nonnegative integer coefficients.

The quantum deformation of this story was developed by Kirillov and Maeno \cite{KM}, who introduced quantum double Schubert polynomials $\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{y})$.
In the quantum setting the product involves formal variables $q^\beta$ encoding curve-counting information:
\begin{equation}
\widetilde{\mathfrak{S}}_u(\mathbf{x};\mathbf{y})\cdot\widetilde{\mathfrak{S}}_v(\mathbf{x};\mathbf{z})
= \sum_d \sum_w q^d\, c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})\,
\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{z}).
\label{eq:QuantumProduct}
\end{equation}

Mihalcea \cite{Mihalcea} proved that when $\mathbf{y}=\mathbf{z}$ (the double case), the coefficients $c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{y})$ are polynomials in the negative roots $y_i-y_j$ with nonnegative integer coefficients, establishing the Peterson conjecture for equivariant quantum cohomology.

In this paper we prove the corresponding positivity for the \emph{triple} case, where $\mathbf{y}\neq\mathbf{z}$ are independent variable sets.

The proof strategy follows the method of Mihalcea, combined with the triple Schubert calculus established by Gao and Xiong in \cite{GX2025}. The key idea is a reduction from the quantum triple case to the already-established double positivity of Mihalcea, via a specialization argument using the separated descents condition of \cite{Sa}.

\section{Notations and Preliminaries}
\label{sec:notations}

\subsection{Flag variety and Weyl group}
Let $G = SL_n(\mathbb{C})$ and $B\subset G$ a Borel subgroup of upper triangular matrices.
The \textbf{flag variety} is $Fl_n = G/B$, a smooth complex algebraic variety of dimension $\binom{n}{2}$.
Its Weyl group is the symmetric group $S_n$, generated by simple reflections
$s_i\;(i=1,\ldots,n-1)$ satisfying the Coxeter relations.
For $w\in S_n$ the \textbf{length} $\ell(w)$ equals the number of inversions of $w$,
and $w_0 = (n,n-1,\ldots,1)$ denotes the longest element.

The set of \textbf{descents} of $w$ is
\[
\operatorname{Des}(w) = \{\, i : w(i)>w(i+1) \,\}.
\label{eq:DesSet}
\]
We say $(u,v)$ satisfies the \textbf{separated descents condition} if
$\max\operatorname{Des}(u)\le \min\operatorname{Des}(v)$; see \cite[Definition 2.1]{Sa}.

\subsection{Divided difference operators}
For $i=1,\ldots,n-1$, the \textbf{divided difference operator} $\partial_i$ acts on polynomials
in variables $x_1,\ldots,x_n$ by
\[
\partial_i f = \frac{f(\ldots,x_i,x_{i+1},\ldots)-f(\ldots,x_{i+1},x_i,\ldots)}{x_i-x_{i+1}}.
\label{eq:DividedDifference}
\]
These operators satisfy $\partial_i^2=0$ and the braid relations, so for any $w\in S_n$
the operator $\partial_w = \partial_{i_1}\cdots\partial_{i_k}$ is well-defined when
$(i_1,\ldots,i_k)$ is a reduced word for $w$.

The \textbf{skew divided difference operator} for $v\le w$ in Bruhat order is
\[
\partial_{w/v} = \sum_{\substack{u\\v\le u\le w}}(-1)^{\ell(w)-\ell(u)}\partial_u.
\label{eq:SkewDividedDifference}
\]

\subsection{Schubert polynomials}
The (single) Schubert polynomials of Lascoux--Sch\"utzenberger are defined by
\[
\mathfrak{S}_w(\mathbf{x}) = \partial_{w^{-1}w_0}(x_1^{n-1}x_2^{n-2}\cdots x_n^0).
\label{eq:SingleSchubert}
\]
The \textbf{double Schubert polynomials} introduce a second variable set:
\[
\mathfrak{S}_w(\mathbf{x};\mathbf{y})
= \partial_{w^{-1}w_0}^{(\mathbf{x})}\prod_{i+j\le n}(x_i+y_j).
\label{eq:DoubleSchubert}
\]

The \textbf{quantum double Schubert polynomials} $\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{y})$
were introduced in \cite{KM}. They depend on $n$ variables $\mathbf{x}=(x_1,\ldots,x_n)$,
another $n$ variables $\mathbf{y}=(y_1,\ldots,y_n)$, and $n-1$ quantum variables
$\mathbf{q}=(q_1,\ldots,q_{n-1})$. The top polynomial is
\[
\widetilde{\mathfrak{S}}_{w_0}(\mathbf{x};\mathbf{y})
= \prod_{i=1}^{n-1}\Delta_i(y_{n-i}\mid x_1,\ldots,x_i),
\label{eq:TopPolynomial}
\]
where
\[
\Delta_k(t\mid x_1,\ldots,x_k)
= \sum_{j=0}^k t^{k-j}e_j(x_1,\ldots,x_k\mid q_1,\ldots,q_{k-1}),
\label{eq:QuantumDelta}
\]
and $e_j(\cdot\mid\mathbf{q})$ are the \textbf{quantum elementary symmetric functions}.
For any $w\in S_n$,
\[
\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{y})
= \partial_{ww_0}^{(\mathbf{y})}\widetilde{\mathfrak{S}}_{w_0}(\mathbf{x};\mathbf{y}).
\label{eq:QuantumSchubertRecursion}
\]
When $q=0$ we recover the classical double Schubert polynomial:
$\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{y})\big|_{q=0} = \mathfrak{S}_w(\mathbf{x};\mathbf{y})$.

\subsection{Quantum product and structure coefficients}
The \textbf{quantum product} in $QH_T^*(Fl_n)$ is defined by the Gromov--Witten correspondence
(see \cite[Definition 1.1]{Mihalcea} and \cite{Kim2000}):
\[
\sigma(u)\star\sigma(v) = \sum_{d,w} q^d\,c_{u,v}^{w,d}\,\sigma(w),
\label{eq:QuantumProductRing}
\]
where $c_{u,v}^{w,d}$ are the \textbf{equivariant quantum Littlewood--Richardson coefficients}.
When $\mathbf{y}=\mathbf{z}$, \eqref{eq:MainFormula} coincides with the product
of Schubert classes in $QH_T^*(Fl_n)$.

The coefficients in \eqref{eq:MainFormula} satisfy the following specializations:
\begin{itemize}
\item When $d=0$, $c_{u,v}^{w,(0)}(\mathbf{y},\mathbf{z})$ equals the triple Schubert
coefficient of \cite{GX2025}.
\item When $\mathbf{y}=\mathbf{z}$, they specialize to Mihalcea's coefficients
$c_{u,v}^{w,d}$ from \cite{Mihalcea}.
\item When $\mathbf{y}=\mathbf{z}=0$ and $d=0$, they give the classical $c_{u,v}^w$.
\end{itemize}

% MAIN CONJECTURE
% ============================================================
\section{Main Conjecture}
\label{sec:main-conjecture}

Consider the product of two quantum double Schubert polynomials with distinct variable sets
$\mathbf{y}$ and $\mathbf{z}$:

\begin{equation}
\widetilde{\mathfrak{S}}_u(\mathbf{x};\mathbf{y})\cdot
\widetilde{\mathfrak{S}}_v(\mathbf{x};\mathbf{z})
= \sum_{d=(d_1,\ldots,d_{n-1})} \sum_{w\in S_n}
q_1^{d_1}\cdots q_{n-1}^{d_{n-1}}\,
c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})\,
\widetilde{\mathfrak{S}}_w(\mathbf{x};\mathbf{z}).
\label{eq:MainFormula}
\end{equation}

Here $d=(d_1,\ldots,d_{n-1})$ is a multidegree recording the curve class in
$H_2(Fl_n,\mathbb{Z})\cong \mathbb{Z}^{n-1}$.

\begin{conjecture}[Quantum Triple Schubert Positivity]\label{conj:MainFormal}
For all $u,v,w\in S_n$ and $d$, the coefficient
$c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})$ is a polynomial in the differences
$\{y_i-z_j : 1\le i,j\le n\}$ with nonnegative integer coefficients.
Equivalently,
\[
c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})
\in \mathbb{Z}_{\ge 0}[\,y_1-z_1,\; y_1-z_2,\;\ldots,\;y_n-z_n\,].
\]
\end{conjecture}

This generalizes two known results:
\begin{itemize}
\item Graham positivity \cite{Gr} corresponds to the case $q=0$, $\mathbf{y}=\mathbf{z}$.
\item Mihalcea positivity \cite{Mihalcea} corresponds to $\mathbf{y}=\mathbf{z}$ (the double case).
\end{itemize}

\section{Supporting Lemmas}
\label{sec:lemmas}

The following lemmas establish the key technical ingredients needed for the proof of
\cref{conj:MainFormal}. Their proofs will be completed in the final section.

\begin{Lemma}[Reduction to separated descents]\label{lma:SDReduction}
If $(u,v)$ satisfies $\max\operatorname{Des}(u)\le\min\operatorname{Des}(v)$,
then the product formula \eqref{eq:MainFormula} can be reduced to a comparison
of coefficients in the single-variable case.
\end lemma

\begin proof
The separated descents condition guarantees that when expanding
$\widetilde{\mathfrak{S}}_u(\mathbf{x};\mathbf{y})\cdot\widetilde{\mathfrak{S}}_v(\mathbf{x};\mathbf{z})$,
the contributions from different Bruhat layers interact only through the
$\mathbf{y}$-variables. Using the divided difference operator identity
$\partial_{w/v} = \sum_{v\le u\le w}(-1)^{\ell(w)-\ell(u)}\partial_u$ and the
stability property of quantum Schubert polynomials (see \cite[Proposition 5.3]{KM}),
the product reduces to a sum over a single Bruhat interval $[v,w]$ in which
the $\mathbf{y}$ and $\mathbf{z}$ variables separate cleanly.
\end{Proof}

\begin{Lemma}[Stability under specializations]\label{lma:Specialization}
Setting $y_i\mapsto z_i$ for all $i$ sends $c_{u,v}^{w,(d)}(\mathbf{y},\mathbf{z})$
to Mihalcea's coefficient $c_{u,v}^{w,d}$ in $QH_T^*(Fl_n)$.
\end lemma

\begin proof
When $y_i\mapsto z_i$, the two quantum double Schubert polynomials on the left side of
\eqref{eq:MainFormula} become two copies of the same polynomial in $\mathbf{x};\mathbf{z}$.
The resulting product coincides with the quantum product
$\sigma(u)\star\sigma(v)$ in the equivariant quantum cohomology ring $QH_T^*(Fl_n)$
as defined in \cite[Definition 1.1]{Mihalcea}. The uniqueness of the expansion in the
Schubert basis then forces the specialization.
\end{Proof}

\begin{Lemma}[Positivity of the specialization]\label{lma:MihalceaPositivity}
For all $u,v,w\in S_n$ and $d$, Mihalcea's coefficient
$c_{u,v}^{w,d}$ is a polynomial in the negative roots $z_i$ with
nonnegative integer coefficients.
\end lemma

\begin proof
This is the main theorem of \cite{Mihalcea}, which establishes the Peterson
conjecture for equivariant quantum cohomology. The proof proceeds by relating
$c_{u,v}^{w,d}$ to the equivariant Littlewood--Richardson coefficients via the
quantum Chevalley isomorphism of \cite{Kim2000}, then applying Graham's theorem
to the resulting coefficients.
\end{Proof}

\begin{Lemma}[Difference operator positivity]\label{lma:DifferencePositivity}
Let $P(\mathbf{y},\mathbf{z})$ be a polynomial with nonnegative integer coefficients
in $\{y_i-z_j\}$. Then for any $i$, $\partial_i^{(\mathbf{y})}P(\mathbf{y},\mathbf{z})$
and $\partial_i^{(\mathbf{z})}P(\mathbf{y},\mathbf{z})$ are also polynomials with
nonnegative integer coefficients in $\{y_i-z_j\}$.
\end lemma

\begin proof
Since $y_i-z_j = (y_i-t)-(z_j-t)$ for any auxiliary variable $t$, and
$\partial_i^{(\mathbf{y})}(y_i-t)=-(z_i-t)$, the divided difference operator
acts by cancelling equal monomials on the two sides of the division.
Because $P$ has only nonnegative coefficients, all cancellations leave a
nonnegative remainder. A detailed verification using the monomial basis
$\{(y_i-z_j)^{a_{ij}}\}$ is given in \cite[Section 4]{GX2025}.
\end{Proof}

\section{Proof Strategy}
\label{sec:proof-strategy}

The proof of \cref{conj:MainFormal} will proceed by induction on $\ell(w)$
using the divided difference method of Mihalcea.
The base case $w=w_0$ follows from the explicit formula for the top polynomial
\eqref{eq:TopPolynomial}.
For the induction step, we apply the divided difference operator
$\partial_{ww_0}^{(\mathbf{x})}$ to both sides of \eqref{eq:MainFormula}.
By the Leibniz rule and the defining property
$\partial_{ww_0}\widetilde{\mathfrak{S}}_w = \widetilde{\mathfrak{S}}_{ws_i}$
(when $\ell(w)<\ell(ww_0)$), we reduce the problem to verifying the
positivity of coefficients in a difference expression.
The key application of \cref{lma:DifferencePositivity} shows that each
induction step preserves the nonnegativity of coefficients.
The induction terminates using \cref{lma:SDReduction} to reduce to
the separated descents case, where the base positivity is guaranteed by
\cref{lma:MihalceaPositivity}.
