% Slide 2: Motivation - 为什么需要 Gauss-Markov？

\begin{frame}{为什么需要 Gauss-Markov？}
\begin{block}{核心问题}
\begin{itemize}
  \item OLS 估计凭什么"最好"？
  \item 有没有比 OLS 更好的估计？
  \item 第 3 章 OLS 是纯代数运算，没有随机假设谈不上"最优"
\end{itemize}
\end{block}
\vspace{1em}
\textbf{直觉说明：} Without any stochastic assumptions, the OLS in Chapter 3 is purely algebraic. If we want to discuss the statistical properties of OLS, we must invoke some statistical modeling assumptions.
\end{frame}

% ============================================================
% Slide 3: 从代数到统计
% ============================================================
\begin{frame}{从代数到统计：引入随机假设的必要性}
\begin{block}{Gauss-Markov 模型}
Gauss-Markov 模型 Assumption \S 4.1 给出随机假设，使得讨论估计量的均值、方差、协方差成为可能。
\end{block}
\vspace{1em}
\begin{alertblock}{关键洞察}
第 3 章 OLS 是纯代数运算 $\Rightarrow$ 没有随机假设谈不上"最优"
\end{alertblock}
\end{frame}

% ============================================================
% Slide 5: Gauss-Markov 模型假设
% ============================================================
\begin{frame}{Gauss-Markov 模型假设}
\begin{block}{Assumption (C1): 线性形式}
\[
y_i = x_i^\top \beta + \varepsilon_i, \quad i = 1, \ldots, n
\]
其中 $x_i$ 是 $p$ 维回归量，$\varepsilon_i$ 是误差项。
\end{block}
\vspace{0.5em}
\textbf{关键条件：}
\begin{itemize}
  \item $X$ 固定且列线性无关
  \item $E(\varepsilon_i) = 0$
  \item $\cov(\varepsilon_i, \varepsilon_j) = \sigma^2 \mathbf{1}\{i=j\}$（同方差、不相关）
\end{itemize}
\end{frame}

% ============================================================
% Slide 6: 个体水平视角
% ============================================================
\begin{frame}{个体水平视角：$y_i = x_i^\top \beta + \varepsilon_i$}
\begin{block}{Assumption 的直观理解}
\begin{itemize}
  \item 每个观测的误差均值 0：$E(\varepsilon_i) = 0$
  \item 每个误差方差 $\sigma^2$（同方差）：$\var(\varepsilon_i) = \sigma^2$
  \item 不同观测间误差不相关：$\cov(\varepsilon_i, \varepsilon_j) = 0, i \neq j$
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 7: 同方差假设
% ============================================================
\begin{frame}{同方差假设的含义与重要性}
\begin{alertblock}{Homoskedasticity（同方差）}
\begin{itemize}
  \item homoskedasticity = 相同方差
  \item 词源学：homo 更好地表示 variance 含义
  \item 异方差情形需要加权最小二乘（见第 19 章）
\end{itemize}
\end{alertblock}
\end{frame}

% ============================================================
% Slide 8: OLS 估计量矩阵形式
% ============================================================
\begin{frame}{OLS 估计量：矩阵形式}
\begin{block}{OLS 估计量}
\[
\hat{\beta} = (X^\top X)^{-1} X^\top Y
\]
\end{block}
\vspace{1em}
\textbf{关键性质：}
\begin{itemize}
  \item $\hat{\beta}$ 是 $Y$ 的线性函数
  \item 仅依赖矩阵运算
  \item $A = (X^\top X)^{-1} X^\top$ 不依赖 $Y$
\end{itemize}
\end{frame}

% ============================================================
% Slide 9: OLS 估计量的均值与方差 (Theorem 4.1)
% ============================================================
\begin{frame}{OLS 估计量的均值与方差 (Theorem 4.1)}
\begin{block}{Theorem 4.1}
在 Gauss-Markov 假设下：
\[
E(\hat{\beta}) = \beta \quad \text{（无偏性）}
\]
\[
\cov(\hat{\beta}) = \sigma^2 (X^\top X)^{-1}
\]
\end{block}
\vspace{0.5em}
\textbf{证明思路：} 利用 $E(Y) = X\beta$ 和 $\cov(Y) = \sigma^2 I_n$
\end{frame}

% ============================================================
% Slide 10: Gauss-Markov 定理（核心）
% ============================================================
\begin{frame}{Gauss-Markov 定理（核心）}
\begin{block}{Theorem 4.2 (Gauss-Markov)}
$\hat{\beta}$ 是 $\beta$ 的 BLUE (Best Linear Unbiased Estimator)：
\begin{itemize}
  \item[(C1)] $\tilde{\beta} = AY$ 对 $Y$ 线性
  \item[(C2)] $E(\tilde{\beta}) = \beta$（对所有 $\beta$ 无偏）
  \item[$\Rightarrow$] $\cov(\tilde{\beta}) \succeq \cov(\hat{\beta})$（正定半定序）
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 11: BLUE 的含义
% ============================================================
\begin{frame}{BLUE 的含义：协方差矩阵序}
\begin{block}{正定半定序 $\cov(\tilde{\beta}) \succeq \cov(\hat{\beta})$}
\begin{itemize}
  \item 对任意线性组合 $c^\top \tilde{\beta}$ 方差不小于 $c^\top \hat{\beta}$
  \item 特别地：每个坐标分量 $\hat{\beta}_j$ 方差最小
  \item 等价形式：$\var(c^\top \tilde{\beta}) \geq \var(c^\top \hat{\beta})$
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 12: 为什么只比较线性估计量？
% ============================================================
\begin{frame}{为什么只比较线性估计量？}
\begin{block}{"线性"约束的合理性}
\begin{itemize}
  \item $A$ 可以是 $X$ 的任意非线性函数
  \item 线性约束已经包含极广的估计量类
  \item 无偏性是自然要求
  \item 在许多现代应用中，有偏估计（如 Ridge、Lasso）方差更小
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 13: OLS 投影几何直观
% ============================================================
\begin{frame}{OLS 投影几何直观}
\begin{block}{投影矩阵}
\[
H = X(X^\top X)^{-1}X^\top
\]
\end{block}
\vspace{0.5em}
\textbf{几何含义：}
\begin{itemize}
  \item $\hat{Y} = HY$ 是 $Y$ 到 $X$ 列空间的投影
  \item 残差 $\hat{\varepsilon} = (I-H)Y$ 与列空间正交
  \item OLS 将 $Y$ 分解为拟合值 + 残差
\end{itemize}
\end{frame}

% ============================================================
% Slide 14: 拟合值与残差的分布性质 (Theorem 4.2)
% ============================================================
\begin{frame}{拟合值与残差的分布性质 (Theorem 4.2)}
\begin{block}{Theorem 4.2}
\[
E(\hat{Y}) = X\beta, \quad E(\hat{\varepsilon}) = 0
\]
\[
\cov(\hat{Y}) = \sigma^2 H, \quad \cov(\hat{\varepsilon}) = \sigma^2(I-H)
\]
$\hat{Y}$ 与 $\hat{\varepsilon}$ 不相关。
\end{block}
\end{frame}

% ============================================================
% Slide 15: 正交 vs. 不相关
% ============================================================
\begin{frame}{正交 vs. 不相关：两个关键陈述的区别}
\begin{alertblock}{常见误区}
\begin{itemize}
  \item \textbf{陈述 (S1)：} $\hat{Y}$ 与 $\hat{\varepsilon}$ 正交 --- 代数事实，无需随机假设
  \item \textbf{陈述 (S2)：} $\hat{Y}$ 与 $\hat{\varepsilon}$ 不相关 --- 随机陈述，需要 Gauss-Markov 假设
\end{itemize}
两者含义不同，不能混淆。
\end{alertblock}
\end{frame}

% ============================================================
% Slide 16: 误差椭圆可视化
% ============================================================
\begin{frame}{误差椭圆可视化："最佳"的含义}
\begin{block}{二维情形}
\begin{itemize}
  \item $\hat{\beta}$ 的置信椭圆
  \item 其他线性无偏估计的置信椭圆
  \item OLS 的置信椭圆最小（被其他所有椭圆包裹）
  \item $\Rightarrow$ $\hat{\beta}$ 在所有方向上方差最小
\end{itemize}
\end{block}
\end{frame}

% ============================================================
% Slide 17: Gauss-Markov 定理证明思路
% ============================================================
\begin{frame}{Gauss-Markov 定理证明思路}
\begin{block}{四步框架}
\begin{enumerate}
  \item 第一步：OLS 满足无偏性条件 $\Rightarrow AX = I_p$
  \item 第二步：协方差分解 $\cov(\tilde{\beta}) = \cov(\hat{\beta}) + \cov(\tilde{\beta} - \hat{\beta})$
  \item 第三步：交叉协方差项为零
  \item 第四步：$\cov(\tilde{\beta} - \hat{\beta}) \succeq 0 \Rightarrow \cov(\tilde{\beta}) \succeq \cov(\hat{\beta})$
\end{enumerate}
\end{block}
\end{frame}

% ============================================================
% Slide 18: 例子：简单线性回归的 BLUE
% ============================================================
\begin{frame}{例子：简单线性回归的 BLUE}
\begin{exampleblock}{Example}
\begin{itemize}
  \item $y_i = \alpha + \beta x_i + \varepsilon_i$
  \item $\var(\hat{\beta}) = \sigma^2 / \sum (x_i - \bar{x})^2$
  \item 任何线性无偏估计的 $\beta$ 系数方差都不小于此值
\end{itemize}
\end{exampleblock}
\end{frame}

% ============================================================
% Slide 19: 例子：均值的 BLUE
% ============================================================
\begin{frame}{例子：均值的 BLUE}
\begin{exampleblock}{Example}
\begin{itemize}
  \item $y_i \sim$ 均值 $\mu$，方差 $\sigma^2$，互不相关
  \item 线性估计 $\hat{\mu} = \sum a_i y_i$，无偏要求 $\sum a_i = 1$
  \item 最优选择：$a_i = 1/n$（简单平均）
  \item 方差 $\var(\hat{\mu}) = \sigma^2/n$ 为最小
\end{itemize}
\end{exampleblock}
\end{frame}

% ============================================================
% Slide 20: 例子：Gauss-Markov 预测定理
% ============================================================
\begin{frame}{例子：Gauss-Markov 预测定理}
\begin{exampleblock}{Example}
\begin{itemize}
  \item $\hat{Y} = X\hat{\beta}$ 是 $X\beta$ 的最佳线性无偏预测
  \item 适用于任何线性预测 $\tilde{Y} = \tilde{H}Y$
  \item $\cov(\tilde{Y}) \succeq \cov(\hat{Y})$
\end{itemize}
\end{exampleblock}
\end{frame}

% ============================================================
% Slide 21: 总结
% ============================================================
\begin{frame}{Gauss-Markov 定理总结}
\begin{block}{主要结论}
\begin{itemize}
  \item Gauss-Markov 模型：线性 + 同方差 + 不相关
  \item OLS $\hat{\beta}$ 是 BLUE
  \item 核心：$\cov(\tilde{\beta}) \succeq \cov(\hat{\beta})$
\end{itemize}
\end{block}
\vspace{1em}
\begin{alertblock}{局限}
\begin{itemize}
  \item 不谈非线性估计
  \item 不谈正态假设
  \item 不谈稳健性
\end{itemize}
\end{alertblock}
\end{frame}
